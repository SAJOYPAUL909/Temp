bind_volumes = []

if mounts:

    create_and_send_volume_tars(
        ssh1,
        VM1_SUDO_PASSWORD,
        ssh2,
        VM2_SUDO_PASSWORD,
        container_name,
        mounts,
        temp_dir1,
        temp_dir2
    )

    container_volume_root = (
        f"{DESTINATION_VOLUME_ROOT}/{container_name}"
    )

    for index, m in enumerate(mounts):

        if m.get("Type") != "bind":
            continue

        source = m.get("Source")
        destination = m.get("Destination")

        if not source or should_skip_path(source):
            continue

        mount_name = os.path.basename(
            source.rstrip("/")
        )

        if not mount_name:
            mount_name = f"mount_{index + 1}"

        bind_dest = (
            f"{container_volume_root}/{mount_name}"
        )

        bind_volumes.append(
            f"-v '{bind_dest}:{destination}:z'"
        )
